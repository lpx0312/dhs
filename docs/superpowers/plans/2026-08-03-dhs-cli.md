> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a cross-platform (Windows/Linux) Go CLI `dhs` that searches Docker Hub repositories and queries image tags, with proxy support via flag + config file + env var.

**Architecture:** Single binary using cobra for CLI, yaml.v3 for config, fatih/color for output, and standard net/http for the Docker Hub API client. Three-layer split: `cmd/` (cobra commands), `internal/dockerhub/` (API client + types), `internal/config/` + `internal/ui/` (supporting). Two subcommands: `search` (repo search, single page) and `tags` (tag query with pagination, plus `--arch` filter and `--gen-pull`).

**Tech Stack:** Go 1.26, github.com/spf13/cobra, gopkg.in/yaml.v3, github.com/fatih/color, standard library net/http + net/url + testing/httptest

**Module name:** `dhs` (local module; `go mod init dhs`)

```go
package config

import (
	"os"
	"path/filepath"

	"gopkg.in/yaml.v3"
)

// Config holds the persistent configuration loaded from file.
// Flag values are passed separately to EffectiveProxy/EffectiveNoColor
// so merge priority stays explicit and testable.
type Config struct {
	Proxy   string `yaml:"proxy"`
	NoColor bool   `yaml:"no_color"`
}

// Load reads a YAML config file. A missing file is not an error:
// a zero-value Config is returned so flag/env fallbacks can apply.
func Load(path string) (*Config, error) {
	cfg := &Config{}
	if path == "" {
		return cfg, nil
	}
	data, err := os.ReadFile(path)
	if err != nil {
		if os.IsNotExist(err) {
			return cfg, nil
		}
		return nil, err
	}
	if err := yaml.Unmarshal(data, cfg); err != nil {
		return nil, err
	}
	return cfg, nil
}

// EffectiveProxy returns the proxy to use, applying priority:
// flag > file > HTTPS_PROXY env > HTTP_PROXY env > "" (direct).
func (c *Config) EffectiveProxy(flagProxy string, _ string) string {
	if flagProxy != "" {
		return flagProxy
	}
	if c.Proxy != "" {
		return c.Proxy
	}
	if v := os.Getenv("HTTPS_PROXY"); v != "" {
		return v
	}
	if v := os.Getenv("HTTP_PROXY"); v != "" {
		return v
	}
	return ""
}

// EffectiveNoColor returns whether color should be disabled,
// applying priority: flag > file.
func (c *Config) EffectiveNoColor(flagNoColor bool) bool {
	if flagNoColor {
		return true
	}
	return c.NoColor
}

// DefaultConfigPath returns the platform-standard config path:
// os.UserConfigDir() + /dhs/config.yaml.
func DefaultConfigPath() (string, error) {
	dir, err := os.UserConfigDir()
	if err != nil {
		return "", err
	}
	return filepath.Join(dir, "dhs", "config.yaml"), nil
}
```

```go
package dockerhub

import (
	"fmt"
	"net/http"
	"net/url"
	"time"
)

const baseURL = "https://hub.docker.com"

// NewClient builds an *http.Client. If proxyURL is empty, the transport
// uses ProxyFromEnvironment (reads HTTPS_PROXY/HTTP_PROXY), matching the
// ps1 script's reliance on env vars. If proxyURL is set, an explicit
// proxy is wired in and env vars are ignored to avoid ambiguity.
func NewClient(proxyURL string) (*http.Client, error) {
	transport := &http.Transport{
		MaxIdleConns:        10,
		IdleConnTimeout:     30 * time.Second,
		TLSHandshakeTimeout: 15 * time.Second,
	}
	if proxyURL != "" {
		u, err := url.Parse(proxyURL)
		if err != nil {
			return nil, fmt.Errorf("invalid proxy URL %q: %w", proxyURL, err)
		}
		if u.Scheme == "" {
			return nil, fmt.Errorf("invalid proxy URL %q: missing scheme", proxyURL)
		}
		transport.Proxy = http.ProxyURL(u)
	} else {
		transport.Proxy = http.ProxyFromEnvironment
	}
	return &http.Client{
		Transport: transport,
		Timeout:   30 * time.Second,
	}, nil
}
```

```go
package dockerhub

// imageInfo is one platform variant under a tag.
type imageInfo struct {
	Architecture string `json:"architecture"`
	Variant      string `json:"variant"`
	OS           string `json:"os"`
}

// tagResult is one tag entry from the /tags/ endpoint.
type tagResult struct {
	Name        string      `json:"name"`
	TagStatus   string      `json:"tag_status"`
	FullSize    int64       `json:"full_size"`
	LastUpdated string      `json:"last_updated"`
	Images      []imageInfo `json:"images"`
}

// paginatedTags is the raw /tags/ response envelope.
type paginatedTags struct {
	Count    int         `json:"count"`
	Next     string      `json:"next"`
	Previous string      `json:"previous"`
	Results  []tagResult `json:"results"`
}

// repoResult is one repository from the /search/repositories/ endpoint.
type repoResult struct {
	Name           string `json:"name"`
	StarCount      int64  `json:"star_count"`
	PullCount      int64  `json:"pull_count"`
	IsOfficial     bool   `json:"is_official"`
	ShortDescription string `json:"short_description"`
	LastUpdated    string `json:"last_updated"`
}

// paginatedRepos is the raw /search/repositories/ response envelope.
type paginatedRepos struct {
	Count   int          `json:"count"`
	Next    string       `json:"next"`
	Results []repoResult `json:"results"`
}

// --- Public, normalized types returned to callers ---

// Tag is the normalized tag info surfaced to the UI layer.

	Name        string
	Status      string
	SizeMB      float64
	Arches      []string // display list (linux-only by default, or os/arch with ShowOS)
	LastUpdated string
	// archSet is the set of linux architecture fields (no variant) for filtering.
	archSet map[string]bool
}

// Repo is the normalized repository info surfaced to the UI layer.

	Name        string
	Stars       int64
	Pulls       int64

}
```

```go
package dockerhub

import (
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"net/url"
)

// FetchRepositories searches Docker Hub repositories for the keyword.
// It performs a single request with page_size=limit (search is ranked;
// top-N paging is sufficient and matches the design spec).
// Returns the normalized repos and the total count reported by the API.
func FetchRepositories(client *http.Client, base, keyword string, limit int) ([]Repo, int, error) {
	if limit <= 0 {
		limit = 25
	}
	u := base + "/v2/search/repositories/?query=" + url.QueryEscape(keyword) +
		"&page_size=" + fmt.Sprintf("%d", limit)

	resp, err := client.Get(u)
	if err != nil {
		return nil, 0, fmt.Errorf("search request failed: %w", err)
	}
	defer resp.Body.Close()

	if resp.StatusCode < 200 || resp.StatusCode >= 300 {
		return nil, 0, fmt.Errorf("docker hub search returned HTTP %d", resp.StatusCode)
	}

	body, err := io.ReadAll(resp.Body)
	if err != nil {
		return nil, 0, fmt.Errorf("reading search response: %w", err)
	}

	var pr paginatedRepos
	if err := json.Unmarshal(body, &pr); err != nil {
		return nil, 0, fmt.Errorf("parsing search response: %w", err)
	}

	out := make([]Repo, 0, len(pr.Results))
	for _, r := range pr.Results {
		out = append(out, Repo{
			Name:        r.Name,
			Stars:       r.StarCount,
			Pulls:       r.PullCount,
			Official:    r.IsOfficial,
			Description: r.ShortDescription,
			LastUpdated: r.LastUpdated,
		})
	}
	return out, pr.Count, nil
}
```

```go
package dockerhub

import (
	"encoding/json"
	"fmt"
	"io"
	"net/http"
	"sort"
	"time"
)

// FetchTags retrieves tags for user/image. It pages through results
// (page_size 100) following the API's next cursor until exhausted.
// If limit > 0, fetching stops once at least limit tags are collected.
// Set limit to 0 to fetch all pages. showOS controls whether non-linux
// architectures are surfaced (with an os/ prefix) in the display list.
func FetchTags(client *http.Client, base, user, image string, limit int, showOS bool) ([]Tag, error) {
	u := fmt.Sprintf("%s/v2/repositories/%s/%s/tags/?page_size=100", base, user, image)
	var out []Tag

	for {
		resp, err := client.Get(u)
		if err != nil {
			return nil, fmt.Errorf("tags request failed: %w", err)
		}
		if resp.StatusCode == http.StatusNotFound {
			resp.Body.Close()
			return nil, fmt.Errorf("image %s/%s not found", user, image)
		}
		if resp.StatusCode < 200 || resp.StatusCode >= 300 {
			resp.Body.Close()
			return nil, fmt.Errorf("docker hub tags returned HTTP %d", resp.StatusCode)
		}
		body, err := io.ReadAll(resp.Body)
		resp.Body.Close()
		if err != nil {
			return nil, fmt.Errorf("reading tags response: %w", err)
		}
		var pr paginatedTags
		if err := json.Unmarshal(body, &pr); err != nil {
			return nil, fmt.Errorf("parsing tags response: %w", err)
		}
		for _, tr := range pr.Results {
			arches, archSet := parseTagArches(tr, showOS)
			out = append(out, Tag{
				Name:        tr.Name,
				Status:      tr.TagStatus,
				SizeMB:      float64(tr.FullSize) / 1024.0 / 1024.0,
				Arches:      arches,
				LastUpdated: tr.LastUpdated,
				archSet:     archSet,
			})
		}
		if pr.Next == "" {
			break
		}
		if limit > 0 && len(out) >= limit {
			break
		}
		u = pr.Next
		time.Sleep(tagsPageSleep) // polite rate limit, matches ps1
	}
	return out, nil
}

// parseTagArches builds the display architecture list and the filtering set.
// showOS mirrors ps1's -ShowOS: when true, non-linux arches get an "os/" prefix.
// archSet holds linux architecture fields (without variant) for --arch filtering.
func parseTagArches(tr tagResult, showOS bool) ([]string, map[string]bool) {
	seen := map[string]bool{}
	archSet := map[string]bool{}
	for _, img := range tr.Images {
		if img.Architecture == "" || img.Architecture == "unknown" {
			continue
		}
		os := img.OS
		if os == "" {
			os = "linux"
		}
		arch := img.Architecture
		if img.Variant != "" {
			arch = arch + "/" + img.Variant
		}
		if showOS {
			if os != "linux" {
				arch = os + "/" + arch
			}
			seen[arch] = true
		} else {
			// Default: linux arches only.
			if os == "linux" {
				seen[arch] = true
				archSet[img.Architecture] = true
			}
		}
		// archSet always tracks linux architecture fields for filtering,
		// even when showOS is true (so --arch + --os compose correctly).
		if showOS && os == "linux" {
			archSet[img.Architecture] = true
		}
	}
	out := make([]string, 0, len(seen))
	for a := range seen {
		out = append(out, a)
	}
	sort.Strings(out)
	return out, archSet
}

// HasArch reports whether the tag's linux architecture set contains target.
func (t Tag) HasArch(target string) bool {
	return t.archSet[target]
}
```

```go
// --- Row structs (pre-formatted strings; the caller formats numbers) ---

type RepoRow struct {
	Name     string
	Stars    string
	Pulls    string
	Official string
	Updated  string
}

type TagRow struct {
	Tag     string
	Status  string
	Size    string
	Arch    string
	Updated string
}

// PullSpec is one tag's matched arches, used to build pull commands.
type PullSpec struct {
	Tag    string
	Arches map[string]bool
}

// renderTable builds a column-aligned text table from headers + rows.
// All cells are already formatted strings.
func renderTable(headers []string, rows [][]string) string {
	widths := make([]int, len(headers))
	for i, h := range headers {
		widths[i] = len(h)
	}
	for _, r := range rows {
		for i := 0; i < len(headers) && i < len(r); i++ {
			if len(r[i]) > widths[i] {
				widths[i] = len(r[i])
			}
		}
	}
	var b strings.Builder
	writeRow := func(cells []string) {
		for i, c := range cells {
			b.WriteString(c)
			pad := widths[i] - len(c)
			if pad > 0 {
				b.WriteString(strings.Repeat(" ", pad))
			}
			if i < len(cells)-1 {
				b.WriteString("   ")
			}
		}
		b.WriteString("\n")
	}
	writeRow(headers)
	// Separator line.
	for i := range headers {
		b.WriteString(strings.Repeat("-", widths[i]))
		if i < len(headers)-1 {
			b.WriteString("   ")
		}
	}
	b.WriteString("\n")
	for _, r := range rows {
		writeRow(r)
	}
	return b.String()
}

// RenderRepoTable returns the aligned repo table text.
func RenderRepoTable(rows []RepoRow) string {
	headers := []string{"NAME", "STARS", "PULLS", "OFFICIAL", "UPDATED"}
	cells := make([][]string, 0, len(rows))
	for _, r := range rows {
		cells = append(cells, []string{r.Name, r.Stars, r.Pulls, r.Official, r.Updated})
	}
	return renderTable(headers, cells)
}

// RenderTagTable returns the aligned tag table text.
func RenderTagTable(rows []TagRow) string {
	headers := []string{"Tag", "Status", "Size(MB)", "Architecture", "LastUpdated"}
	cells := make([][]string, 0, len(rows))
	for _, r := range rows {
		cells = append(cells, []string{r.Tag, r.Status, r.Size, r.Arch, r.Updated})
	}
	return renderTable(headers, cells)
}

// FormatPullCommands builds docker pull command strings for given specs.
// user=="library" => image name omits the library/ prefix (matches ps1).
// amd64 => plain "<image>:<tag>"; arm64 => "--platform=linux/arm64 <image>:<tag>".
// Output is ordered by tag name then amd64-before-arm64.
func FormatPullCommands(user, image string, specs []PullSpec) []string {
	imageName := image
	if user != "library" {
		imageName = user + "/" + image
	}
	sort.Slice(specs, func(i, j int) bool { return specs[i].Tag < specs[j].Tag })
	var out []string
	for _, s := range specs {
		if s.Arches["amd64"] {
			out = append(out, fmt.Sprintf("%s:%s", imageName, s.Tag))
		}
		if s.Arches["arm64"] {
			out = append(out, fmt.Sprintf("--platform=linux/arm64 %s:%s", imageName, s.Tag))
		}
	}
	return out
}
```

			}
			fmt.Printf("\n%s\n", ui.Green(fmt.Sprintf("标签信息已保存到文件: %s", tagsOutput)))
		}
		return nil
	},
}

// parseImageRef splits "user/image" into (user, image). Without a slash,
// user defaults to "library" (official images).
func parseImageRef(ref string) (string, string) {
	if i := strings.Index(ref, "/"); i >= 0 {
		return ref[:i], ref[i+1:]
	}
	return "library", ref
}

// parseArchFlag turns "amd64,arm64" into ["amd64","arm64"]; empty => nil.
func parseArchFlag(s string) []string {
	s = strings.TrimSpace(s)
	if s == "" {
		return nil
	}
	parts := strings.Split(s, ",")
	out := make([]string, 0, len(parts))
	for _, p := range parts {
		if p = strings.TrimSpace(p); p != "" {
			out = append(out, p)
		}
	}
	return out
}

func filterByName(tags []dockerhub.Tag, sub string) []dockerhub.Tag {
	out := make([]dockerhub.Tag, 0, len(tags))
	for _, t := range tags {
		if strings.Contains(t.Name, sub) {
			out = append(out, t)
		}
	}
	return out
}

func filterByArch(tags []dockerhub.Tag, targets []string) []dockerhub.Tag {
	set := map[string]bool{}
	for _, a := range targets {
		set[a] = true
	}
	out := make([]dockerhub.Tag, 0, len(tags))
	for _, t := range tags {
		for a := range set {
			if t.HasArch(a) {
				out = append(out, t)
				break
			}
		}
	}
	return out
}

func sortTags(tags []dockerhub.Tag, by string, asc bool) {
	less := func(i, j int) bool {
		a, b := tags[i], tags[j]
		var less bool
		switch by {
		case "name":
			less = a.Name < b.Name
		case "size":
			less = a.SizeMB < b.SizeMB
		default: // date
			less = a.LastUpdated < b.LastUpdated
		}
		if asc {
			return less
		}
		return !less
	}
	sort.SliceStable(tags, less)
}

func renderTagRows(tags []dockerhub.Tag) string {
	rows := make([]ui.TagRow, 0, len(tags))
	for _, t := range tags {
		rows = append(rows, ui.TagRow{
			Tag:     t.Name,
			Status:  t.Status,
			Size:    strconv.FormatFloat(t.SizeMB, 'f', 1, 64),
			Arch:    strings.Join(t.Arches, ", "),
			Updated: shortDate(t.LastUpdated),
		})
	}
	return ui.RenderTagTable(rows)
}

func buildPullSpecs(tags []dockerhub.Tag, targets []string) []ui.PullSpec {
	specs := make([]ui.PullSpec, 0, len(tags))
	for _, t := range tags {
		arches := map[string]bool{}
		for _, a := range targets {
			if t.HasArch(a) {
				arches[a] = true
			}
		}
		if len(arches) > 0 {
			specs = append(specs, ui.PullSpec{Tag: t.Name, Arches: arches})
		}
	}
	return specs
}

// exportTags writes a markdown summary of the displayed tags (and optional
// pull commands) to path.
func exportTags(path, user, name string, tags []dockerhub.Tag, archTargets []string, genPull bool) error {
	var b strings.Builder
	fmt.Fprintf(&b, "# Docker Hub 镜像标签信息\n\n")
	fmt.Fprintf(&b, "- 镜像: `%s/%s`\n", user, name)
	if len(archTargets) > 0 {
		fmt.Fprintf(&b, "- 架构过滤: %s\n", strings.Join(archTargets, ", "))
	}
	fmt.Fprintf(&b, "- 标签数: %d\n\n", len(tags))
	fmt.Fprintf(&b, "| Tag | Status | Size(MB) | Architecture | LastUpdated |\n")
	fmt.Fprintf(&b, "| --- | --- | --- | --- | --- |\n")
	for _, t := range tags {
		fmt.Fprintf(&b, "| %s | %s | %.1f | %s | %s |\n",
			t.Name, t.Status, t.SizeMB, strings.Join(t.Arches, ", "), shortDate(t.LastUpdated))
	}
	if genPull && len(archTargets) > 0 {
		specs := buildPullSpecs(tags, archTargets)
		cmds := ui.FormatPullCommands(user, name, specs)
		fmt.Fprintf(&b, "\n## Docker 镜像拉取命令\n\n")
		fmt.Fprintf(&b, "```\n")
		for _, c := range cmds {
			fmt.Fprintln(&b, c)
		}
		fmt.Fprintf(&b, "```\n")
	}
	return osWriteFile(path, []byte(b.String()), 0644)
}

- **Field names**: The Docker Hub API JSON field names (`name`, `tag_status`, `full_size`, `last_updated`, `images[].architecture|variant|os`, `star_count`, `pull_count`, `is_official`, `short_description`) are taken from the reference `docker_hub_search.ps1` (which already parses them) and the public Docker Hub API. If any field comes back empty during the smoke test, check the actual JSON (`--proxy ... curl https://hub.docker.com/v2/repositories/library/nginx/tags/?page_size=1`) and adjust the struct tags in `internal/dockerhub/types.go`.
- **Proxy precedence**: `NewClient("")` uses `ProxyFromEnvironment`, so even without `--proxy`/config, `HTTPS_PROXY`/`HTTP_PROXY` work — matching the ps1 behavior exactly.
- **Tests are offline**: All `internal/` tests use `httptest`; they never hit the real Docker Hub. The smoke tests in Task 10 are the only network-dependent checks.
- **500ms page delay**: controlled by `tagsPageSleep` so tests run fast; production default is 500ms per the spec.
